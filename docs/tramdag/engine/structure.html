tramdag.engine.structure

Translate the per-node entries of configuration.json into the structure of a node model (which inputs feed the intercept network and which feed each shift network) and encode pandas data into the corresponding design matrices.

The ordering and grouping rules are identical to the original implementation (ordered_parents / group_by_base / GenericDataset) so parameters are interchangeable with the PyTorch version:

  • terms are sorted ci < ciXX < cs < csXX < ls,
  • all ci* terms feed one intercept network (otherwise a simple intercept with a constant input of 1 is used),
  • every plain cs / ls term gets its own shift network, numbered terms such as cs11, cs12 share one network (interaction cs(x1, x2)),
  • parents whose data type contains Xn are one-hot encoded, everything else is fed as a single float column.

 47def _sort_key(term):
 48    base = re.match(r"[a-zA-Z]+", term)
 49    digits = re.findall(r"\d+", term)
 50    base = base.group(0) if base else ""
 51    d = int(digits[0]) if digits else -1
 52    if base in ("ci", "cs") and d != -1:
 53        return (_ORDER.index(base + "XX"), d)
 54    if base in _ORDER:
 55        return (_ORDER.index(base), -1)
 56    return (len(_ORDER), d)

 59def ordered_terms(node_conf):
 60    """Parents sorted like the original ``ordered_parents``: returns OrderedDict parent -> (term, model)."""
 61    terms = node_conf.get("transformation_terms_in_h()", {}) or {}
 62    models = node_conf.get("transformation_term_nn_models_in_h()", {}) or {}
 63    keys = sorted(terms.keys(), key=lambda k: _sort_key(terms[k]))
 64    return OrderedDict((k, (terms[k], models.get(k))) for k in keys)

 67def _group_key(parent, term, prefixes):
 68    for p in prefixes:
 69        if term.startswith(p):
 70            if term == p:
 71                return parent
 72            if len(term) > len(p) and term[len(p)].isdigit():
 73                return term[: len(p) + 1]
 74            return term
 75    return None

125    def formula(self):
126        """Human readable form of the transformation function, e.g. ``h(x3 | x1, x2) = SI + ls(x1) + cs(x2)``."""
127        lhs = f"h({self.name} | {', '.join(self.parents)})" if self.parents else f"h({self.name})"
128        rhs = [f"ci({', '.join(self.intercept.parents)})" if self.intercept else "SI"]
129        rhs += [g.label for g in self.shifts]
130        return f"{lhs} = {' + '.join(rhs)}"

132    # ------------------------------------------------------------------ encoding

134    def _encode_parent(self, parent, values):
135        values = np.asarray(values)
136        if is_onehot_predictor(self.parent_types[parent]):
137            k = self.parent_levels[parent]
138            idx = values.astype(np.int64)
139            if idx.size and (idx.min() < 0 or idx.max() >= k):
140                raise ValueError(f"parent '{parent}' of '{self.name}' has values outside 0..{k-1}")
141            out = np.zeros((len(idx), k), np.float32)
142            out[np.arange(len(idx)), idx] = 1.0
143            return out
144        return values.astype(np.float32).reshape(-1, 1)
145
146    def encode_inputs(self, columns: dict, n: int):
147        """``columns``: mapping parent -> 1d array of length ``n``. Returns ``{"int": ..., "shifts": [...]}``."""
148        if self.intercept is None:
149            int_x = np.ones((n, 1), np.float32)
150        else:
151            int_x = np.concatenate([self._encode_parent(p, columns[p]) for p in self.intercept.parents], axis=1)
152        shifts = [
153            np.concatenate([self._encode_parent(p, columns[p]) for p in g.parents], axis=1) for g in self.shifts
154        ]
155        return {"int": int_x, "shifts": shifts}
156
157    def encode_target(self, values):
158        values = np.asarray(values)
159        if self.target == "continuous":
160            return values.astype(np.float32)
161        y = values.astype(np.int64)
162        if y.size and (y.min() < 0 or y.max() >= self.levels):
163            raise ValueError(f"ordinal target '{self.name}' must take values in 0..{self.levels-1}")
164        return y.astype(np.int32)
165
166    def encode_frame(self, df: pd.DataFrame, with_target=True):
167        missing = [p for p in self.parents if p not in df.columns]
168        if missing:
169            raise KeyError(f"columns {missing} (parents of '{self.name}') missing from data")
170        cols = {p: column_values(df[p]) for p in self.parents}
171        out = self.encode_inputs(cols, len(df))
172        if with_target:
173            if self.name not in df.columns:
174                raise KeyError(f"target column '{self.name}' missing from data")
175            out["y"] = self.encode_target(column_values(df[self.name]))
176        return out

188def build_node_structure(node: str, nodes_conf: dict) -> NodeStructure:
189    conf = nodes_conf[node]
190    dt = conf["data_type"]
191    if is_continuous_target(dt):
192        target, n_thetas, levels = "continuous", DEFAULT_N_THETAS_CONTINUOUS, conf.get("levels")
193    elif is_ordinal_target(dt):
194        levels = conf.get("levels")
195        if levels is None:
196            raise ValueError(
197                f"Ordinal node '{node}' has no 'levels' in the configuration. "
198                "Use cfg.compute_levels(train_df) or TramDagConfig.from_dataframe(...)."
199            )
200        target, n_thetas = "ordinal", int(levels) - 1
201    else:
202        raise NotImplementedError(f"data type '{dt}' of node '{node}' is not supported by the JAX engine")
203
204    parent_types = dict(conf.get("parents_datatype", {}) or {})
205    parent_levels = {}
206    for p, pdt in parent_types.items():
207        if is_onehot_predictor(pdt):
208            lv = nodes_conf.get(p, {}).get("levels")
209            if lv is None:
210                raise ValueError(f"Parent '{p}' of '{node}' is one-hot encoded (Xn) but has no 'levels'.")
211            parent_levels[p] = int(lv)
212
213    def n_feat(parents):
214        return sum(parent_levels.get(p, 1) for p in parents)
215
216    ordered = ordered_terms(conf)
217    int_groups, shift_groups = OrderedDict(), OrderedDict()
218    for parent, (term, model) in ordered.items():
219        if term.startswith(("ci", "si")):
220            key = _group_key(parent, term, ("ci", "si"))
221            int_groups.setdefault(key, []).append((parent, term, model))
222        else:
223            key = _group_key(parent, term, ("cs", "ls"))
224            shift_groups.setdefault(key, []).append((parent, term, model))
225
226    if len(int_groups) > 1:
227        raise ValueError(f"Node '{node}': multiple complex intercept groups detected; only one is supported.")
228    intercept = None
229    for feats in int_groups.values():
230        parents = [f[0] for f in feats]
231        model = feats[0][2] or "ComplexInterceptDefaultTabular"
232        intercept = TermGroup(parents, [f[1] for f in feats], model, n_feat(parents), "intercept")
233
234    shifts = []
235    for feats in shift_groups.values():
236        parents = [f[0] for f in feats]
237        term = feats[0][1]
238        default = "LinearShift" if term.startswith("ls") else "ComplexShiftDefaultTabular"
239        model = feats[0][2] or default
240        shifts.append(TermGroup(parents, [f[1] for f in feats], model, n_feat(parents), "shift"))
241
242    # fail early on unknown model names
243    get_model_spec(intercept.model_name if intercept else "SimpleIntercept")
244    for g in shifts:
245        get_model_spec(g.model_name)
246
247    return NodeStructure(
248        name=node,
249        data_type=dt,
250        target=target,
251        n_thetas=n_thetas,
252        levels=int(levels) if levels is not None else None,
253        parents=list(conf.get("parents", []) or []),
254        intercept=intercept,
255        shifts=shifts,
256        parent_types=parent_types,
257        parent_levels=parent_levels,
258    )

261def topological_order(nodes_conf: dict):
262    """Nodes in an order where every parent precedes its children (stable w.r.t. the config order)."""
263    order, done = [], set()
264    pending = list(nodes_conf.keys())
265    while pending:
266        progressed = False
267        for n in list(pending):
268            if all(p in done for p in nodes_conf[n].get("parents", []) or []):
269                order.append(n)
270                done.add(n)
271                pending.remove(n)
272                progressed = True
273        if not progressed:
274            raise ValueError(f"The graph has a cycle among {pending}")
275    return order
