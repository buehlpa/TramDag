TramDagModel - the user facing model class, running on the JAX engine.

The public methods mirror the original PyTorch class (from_config, from_directory, fit, sample, get_latent, nll, summary and the plot_* diagnostics) and the experiment directory keeps the same layout and JSON files, so existing notebooks keep working. Parameters are stored as .npz files (initial_params.npz, best_params.npz, last_params.npz); legacy *.pt checkpoints are converted on the fly when torch is installed.

  63def _resolve_device(device):
  64    if device in (None, "auto"):
  65        return jax.devices()[0]
  66    if isinstance(device, str):
  67        kind = {"cuda": "gpu"}.get(device.split(":")[0].lower(), device.split(":")[0].lower())
  68        idx = int(device.split(":")[1]) if ":" in device else 0
  69        try:
  70            return jax.devices(kind)[idx]
  71        except RuntimeError:
  72            warnings.warn(f"device '{device}' not available, falling back to {jax.devices()[0]}")
  73            return jax.devices()[0]
  74    return device

 115    @classmethod
 116    def from_config(cls, cfg, **kwargs):
 117        """Build one model per node from a :class:`TramDagConfig`.
 118
 119        Keyword arguments (scalars or ``{node: value}`` dicts where it makes sense):
 120        ``device``, ``seed``, ``verbose``, ``debug``, ``set_initial_weights`` (warm start of
 121        intercepts/linear shifts with R ``tram::Colr`` / ``MASS::polr``), ``initial_data``
 122        (training DataFrame or CSV path for the warm start), ``overwrite_initial_weights``.
 123        """
 124        self = cls()
 125        unknown = set(kwargs) - set(cls.DEFAULTS_CONFIG)
 126        if unknown:
 127            raise ValueError(f"[from_config] Unknown parameter(s): {', '.join(sorted(unknown))}")
 128        settings = {**cls.DEFAULTS_CONFIG, **kwargs}
 129        self.cfg = cfg
 130        if getattr(cfg, "CONF_DICT_PATH", None) and os.path.exists(cfg.CONF_DICT_PATH):
 131            cfg.update()
 132        cfg._verify_completeness()
 133        if getattr(cfg, "CONF_DICT_PATH", None):
 134            try:
 135                cfg.save()
 136            except Exception as e:  # pragma: no cover - best effort like the original
 137                print(f"[WARNING] Could not save configuration: {e}")
 138
 139        self.debug, self.verbose = bool(settings["debug"]), bool(settings["verbose"])
 140        self.device = _resolve_device(settings["device"])
 141        self.nodes_dict = cfg.conf_dict["nodes"]
 142        self.order = topological_order(self.nodes_dict)
 143        self.structures = {n: build_node_structure(n, self.nodes_dict) for n in self.order}
 144        self.settings = settings
 145
 146        key = jax.random.PRNGKey(int(settings["seed"] or 0))
 147        keys = dict(zip(self.order, jax.random.split(key, len(self.order))))
 148        with jax.default_device(self.device):
 149            self.params = {n: init_node_params(keys[n], self.structures[n]) for n in self.order}
 150
 151        set_init = settings["set_initial_weights"]
 152        for n in self.order:
 153            if (set_init[n] if isinstance(set_init, dict) else set_init):
 154                self.params[n] = self._r_warm_start(n, settings["initial_data"])
 155
 156        self.best_params = dict(self.params)
 157        exp_dir = self.experiment_dir
 158        if exp_dir:
 159            for n in self.order:
 160                path = os.path.join(exp_dir, n, _STATE_FILES["init"])
 161                if settings["overwrite_initial_weights"] or not os.path.exists(path):
 162                    save_params(path, self.params[n])
 163        if self.verbose:
 164            for n in self.order:
 165                print(f"[INFO] {self.structures[n].formula()}")
 166        return self

 168    @classmethod
 169    def from_directory(cls, EXPERIMENT_DIR, device="auto", debug=False, verbose=False, state="best"):
 170        """Load configuration, scaling and trained parameters from an experiment directory."""
 171        config_path = os.path.join(EXPERIMENT_DIR, "configuration.json")
 172        if not os.path.exists(config_path):
 173            raise FileNotFoundError(f"[ERROR] Config file not found at {config_path}")
 174        cfg = TramDagConfig.load_json(config_path)
 175        self = cls.from_config(cfg, device=device, debug=debug, verbose=verbose, overwrite_initial_weights=False)
 176        minmax_path = os.path.join(EXPERIMENT_DIR, "min_max_scaling.json")
 177        if not os.path.exists(minmax_path):
 178            raise FileNotFoundError(f"[ERROR] MinMax file not found at {minmax_path}")
 179        with open(minmax_path) as f:
 180            self.minmax_dict = json.load(f)
 181        for n in self.order:
 182            p = self._load_state(n, "last")
 183            if p is not None:
 184                self.params[n] = p
 185            b = self._load_state(n, state)
 186            self.best_params[n] = b if b is not None else self.params[n]
 187        if verbose or debug:
 188            print(f"[INFO] Loaded TramDagModel from {EXPERIMENT_DIR}")
 189        return self

 210    def _minmax(self, node):
 211        if not hasattr(self, "minmax_dict"):
 212            raise RuntimeError(
 213                "[ERROR] minmax_dict not found. Call .fit() or .load_or_compute_minmax(train_df) first."

 217    def _load_state(self, node, state):
 218        """Parameters of ``state`` ('best' | 'last' | 'init') from disk, or None."""
 219        node_dir = self._node_dir(node)
 220        template = self.params[node]
 221        path = os.path.join(node_dir, _STATE_FILES[state])
 222        if os.path.exists(path):
 223            return load_params(path, template)
 224        legacy = os.path.join(node_dir, _LEGACY_FILES[state])
 225        if os.path.exists(legacy):
 226            try:
 227                import torch
 228            except ImportError:
 229                warnings.warn(f"{legacy} is a torch checkpoint; install torch to convert it.")
 230                return None
 231            sd = torch.load(legacy, map_location="cpu")
 232            return params_from_torch_state_dict(sd, template)
 233        return None

 235    def _params_for(self, state):
 236        if state == "best":
 237            return self.best_params
 238        if state == "last":
 239            return self.params
 240        if state == "init":
 241            out = {}
 242            for n in self.order:
 243                p = self._load_state(n, "init") if self.experiment_dir else None
 244                out[n] = p if p is not None else self.params[n]
 245            return out
 246        raise ValueError("state must be one of 'best', 'last', 'init'")

 261    def _r_warm_start(self, node, initial_data):
 262        """Initialise intercepts (and pure linear shifts) from R Colr/POLR estimates."""
 263        from .utils.r_subprocess import fit_r_model_subprocess
 264        import tempfile
 265
 266        if initial_data is None:
 267            raise ValueError("initial_data (DataFrame or CSV path) is required when set_initial_weights=True")
 268        st = self.structures[node]
 269        cleanup = None
 270        if hasattr(initial_data, "to_csv"):
 271            fd, cleanup = tempfile.mkstemp(suffix=".csv")
 272            os.close(fd)
 273            initial_data.to_csv(cleanup, index=False)
 274            path = cleanup
 275        else:
 276            path = initial_data
 277        only_ls = bool(st.shifts) and all(g.is_linear and len(g.parents) == 1 for g in st.shifts)
 278        covs = [g.parents[0] for g in st.shifts] if only_ls else None
 279        try:
 280            thetas_R, shift_coefs = fit_r_model_subprocess(
 281                node, "continous" if st.target == "continuous" else "ordinal", st.n_thetas, path,
 282                ls_shift_covariates=covs, debug=self.debug,
 283            )
 284        finally:
 285            if cleanup and os.path.exists(cleanup):
 286                os.remove(cleanup)
 287        return apply_warm_start(self.params[node], st, thetas_R, shift_coefs if only_ls else None)

 291    def load_or_compute_minmax(self, td_train_data=None, use_existing=False, write=True):
 292        """5% / 95% quantiles of the training data, used to scale continuous targets."""
 293        path = os.path.join(self.experiment_dir, "min_max_scaling.json") if self.experiment_dir else None
 294        if use_existing:
 295            if not path or not os.path.exists(path):
 296                raise FileNotFoundError(f"MinMax file not found: {path}")
 297            with open(path) as f:
 298                self.minmax_dict = json.load(f)
 299            return self.minmax_dict
 300        df = self._as_frame(td_train_data)
 301        num = df.select_dtypes(include="number")  # excludes bool / categorical (only continuous targets are scaled)
 302        q = num.quantile([0.05, 0.95])
 303        self.minmax_dict = {c: [float(q.loc[0.05, c]), float(q.loc[0.95, c])] for c in q.columns}
 304        if write and path:
 305            os.makedirs(self.experiment_dir, exist_ok=True)
 306            with open(path, "w") as f:
 307                json.dump(self.minmax_dict, f, indent=4)
 308        return self.minmax_dict

 312    def fit(self, train_data, val_data=None, **kwargs):
 313        """Train all (or ``train_list``) node models jointly.
 314
 315        Main options: ``epochs``, ``learning_rate``, ``batch_size`` (each a scalar, or for
 316        ``epochs``/``learning_rate`` a ``{node: value}`` dict), ``optimizer`` (an optax
 317        ``GradientTransformation`` replacing Adam), ``seed``, ``verbose``, ``resume`` (continue
 318        from ``last_params.npz`` and the stored histories, default True).
 319        Defaults come from ``cfg.conf_dict['training']`` when present (written by the
 320        configuration editor), otherwise from ``DEFAULTS_FIT``.
 321        Arguments specific to torch DataLoaders (``num_workers`` ...) are accepted and ignored.
 322        """
 323        ignored = set(kwargs) & _IGNORED_FIT_KWARGS
 324        for k in ignored:
 325            kwargs.pop(k)
 326        unknown = set(kwargs) - set(self.DEFAULTS_FIT)
 327        if unknown:
 328            raise ValueError(f"[fit] Unknown parameter(s): {', '.join(sorted(unknown))}")
 329        cfg_training = {k: v for k, v in (self.cfg.conf_dict.get("training") or {}).items() if k in self.DEFAULTS_FIT}
 330        s = {**self.DEFAULTS_FIT, **cfg_training, **kwargs}
 331        self.verbose, self.debug = bool(s["verbose"]), bool(s["debug"])
 332        if "device" in kwargs:
 333            self.device = _resolve_device(s["device"])
 334
 335        train_df = self._as_frame(train_data)
 336        val_df = self._as_frame(val_data)
 337        nodes = [n for n in self.order if n in (s["train_list"] or self.order)]
 338        per_node = lambda v, n: v[n] if isinstance(v, dict) else v
 339
 340        self.load_or_compute_minmax(train_df, use_existing=False, write=True)
 341
 342        # ---- resume
 343        hist_old, start, remaining = {}, {}, {}
 344        for n in nodes:
 345            hist_old[n] = self._read_histories(n) if s["resume"] else None
 346            last = self._load_state(n, "last") if (s["resume"] and self.experiment_dir) else None
 347            if hist_old[n] and last is not None:
 348                self.params[n] = last
 349                best = self._load_state(n, "best")
 350                self.best_params[n] = best if best is not None else last
 351                start[n] = len(hist_old[n]["train"])
 352            else:
 353                hist_old[n] = None
 354                start[n] = 0
 355            remaining[n] = max(0, int(per_node(s["epochs"], n)) - start[n])
 356            if self.verbose:
 357                if remaining[n] == 0:
 358                    print(f"[INFO] Node {n} already trained for {start[n]} epochs. Skipping.")
 359                elif start[n]:
 360                    print(f"[INFO] Node {n}: resuming at epoch {start[n]}.")
 361        nodes = [n for n in nodes if remaining[n] > 0]
 362        if not nodes:
 363            return {} if s["return_history"] else None
 364
 365        if self.verbose:
 366            print(f"[INFO] Training {len(nodes)} node(s) jointly on {self.device} "
 367                  f"({len(train_df)} rows, batch size {s['batch_size']})")
 368            for n in nodes:
 369                print(f"       {self.structures[n].formula()}")
 370
 371        best_score = {}
 372        for n in nodes:
 373            h = hist_old[n]
 374            if h:
 375                vals = [v for v in (h["val"] if any(np.isfinite(h["val"])) else h["train"]) if np.isfinite(v)]
 376                best_score[n] = min(vals) if vals else np.inf
 377            else:
 378                best_score[n] = np.inf
 379
 380        seed = s["seed"] if s["seed"] is not None else int(self.settings.get("seed") or 0) + sum(start.values())
 381        with jax.default_device(self.device):
 382            train_enc = self._encode(train_df, nodes)
 383            val_enc = self._encode(val_df, nodes) if val_df is not None and len(val_df) else None
 384            trainer = JointTrainer(
 385                {n: self.structures[n] for n in nodes}, nodes, {n: self._minmax(n) for n in nodes if self.structures[n].target == "continuous"},
 386                s["batch_size"], {n: per_node(s["learning_rate"], n) for n in nodes}, optimizer=s["optimizer"], seed=seed,
 387            )
 388            p_sub = {n: self.params[n] for n in nodes}
 389            b_sub = {n: self.best_params.get(n, self.params[n]) for n in nodes}
 390            p_sub, b_sub, best_score, hist = trainer.fit(
 391                p_sub, train_enc, val_enc, remaining, best_params=b_sub, best_score=best_score,
 392                verbose=self.verbose, epochs_per_call=s["epochs_per_call"],
 393            )
 394        for n in nodes:
 395            self.params[n] = p_sub[n]
 396            self.best_params[n] = b_sub[n]
 397        self._write_fit_outputs(nodes, hist, hist_old, start, remaining, s)
 398        if self.verbose:
 399            print(f"[INFO] Training finished in {hist['seconds']:.1f}s")
 400        if s["return_history"]:
 401            return {n: (self._read_histories(n) or {}) for n in nodes}
 402        return None

 404    def _read_histories(self, node):
 405        if not self.experiment_dir:
 406            return None
 407        d = self._node_dir(node)
 408        tp, vp = os.path.join(d, "train_loss_hist.json"), os.path.join(d, "val_loss_hist.json")
 409        if not (os.path.exists(tp) and os.path.exists(vp)):
 410            return None
 411        with open(tp) as f:
 412            tr = json.load(f)
 413        with open(vp) as f:
 414            va = json.load(f)
 415        return {"train": [float("nan") if v is None else v for v in tr], "val": [float("nan") if v is None else v for v in va]}

 417    def _write_fit_outputs(self, nodes, hist, hist_old, start, remaining, s):
 418        if not self.experiment_dir:
 419            return
 420        for i, n in enumerate(nodes):
 421            d = self._node_dir(n)
 422            os.makedirs(d, exist_ok=True)
 423            E = remaining[n]
 424            tr = [float(v) for v in hist["train"][:E, i]]
 425            va = [float(v) if np.isfinite(v) else None for v in hist["val"][:E, i]]
 426            if hist_old[n]:
 427                tr = list(hist_old[n]["train"]) + tr
 428                va = [None if (isinstance(v, float) and not np.isfinite(v)) else v for v in hist_old[n]["val"]] + va
 429            with open(os.path.join(d, "train_loss_hist.json"), "w") as f:
 430                json.dump(tr, f)
 431            with open(os.path.join(d, "val_loss_hist.json"), "w") as f:
 432                json.dump(va, f)
 433            save_params(os.path.join(d, _STATE_FILES["last"]), self.params[n])
 434            save_params(os.path.join(d, _STATE_FILES["best"]), self.best_params[n])
 435
 436            st = self.structures[n]
 437            lin = [g for g in st.shifts if g.is_linear]
 438            if s["save_linear_shifts"] and lin:
 439                path = os.path.join(d, "linear_shifts_all_epochs.json")
 440                data = self._json_or_empty(path) if start[n] else {}
 441                for e in range(E):
 442                    data[f"epoch_{start[n] + e + 1}"] = {
 443                        f"ls({', '.join(g.parents)})": _squeeze_list(hist["ls"][n][j][e]) for j, g in enumerate(lin)
 444                    }
 445                with open(path, "w") as f:
 446                    json.dump(data, f)
 447            if s["save_simple_intercepts"] and n in hist["si"]:
 448                path = os.path.join(d, "simple_intercepts_all_epochs.json")
 449                data = self._json_or_empty(path) if start[n] else {}
 450                for e in range(E):
 451                    data[f"epoch_{start[n] + e + 1}"] = [[float(v)] for v in hist["si"][n][e]]
 452                with open(path, "w") as f:
 453                    json.dump(data, f)

 464    def nll(self, data, variables=None, state="best"):
 465        """Mean negative log-likelihood per node on ``data``."""
 466        df = self._as_frame(data)
 467        nodes = [n for n in self.order if variables is None or n in variables]
 468        params = self._params_for(state)
 469        mm = {n: self._minmax(n) for n in nodes if self.structures[n].target == "continuous"}
 470        with jax.default_device(self.device):
 471            fn = make_per_sample_nll(self.structures, nodes, mm)
 472            out = fn({n: params[n] for n in nodes}, self._encode(df, nodes))
 473        return {n: float(jnp.mean(out[n])) for n in nodes}

 475    def get_latent(self, df, verbose=False, state="best"):
 476        """Abduction: ``{node}_U = h(x | pa)`` for continuous nodes, ``{node}_U_lower/_U_upper``
 477        (the latent interval of the observed class) for ordinal nodes."""
 478        params = self._params_for(state)
 479        cols = {}
 480        with jax.default_device(self.device):
 481            for n in self.order:
 482                st = self.structures[n]
 483                enc = self._encode(df, [n])[n]
 484                int_out, shift_sum, _ = node_forward(st, params[n], enc)
 485                cols[n] = df[n].to_numpy()
 486                if st.target == "continuous":
 487                    k_min, k_max = self._minmax(n)
 488                    _, h = continuous_nll_terms(int_out, shift_sum, enc["y"], k_min, k_max)
 489                    cols[f"{n}_U"] = np.asarray(h)
 490                else:
 491                    lo, up = ordinal_bounds(int_out, shift_sum, enc["y"])
 492                    cols[f"{n}_U_lower"] = np.asarray(lo)
 493                    cols[f"{n}_U_upper"] = np.asarray(up)
 494        out = pd.DataFrame(cols, index=df.index)
 495        if verbose:
 496            print("[INFO] Final latent DataFrame shape:", out.shape)
 497        return out

 499    def transformation(self, node, df, x=None, state="best"):
 500        """Evaluate ``h(x | pa)`` for the rows of ``df`` (continuous nodes).
 501
 502        If ``x`` (1d grid) is given, returns an array ``(len(df), len(x))`` with ``h`` evaluated on
 503        the grid for each row's parents; otherwise ``h`` at the observed ``df[node]``.
 504        """
 505        st = self.structures[node]
 506        if st.target != "continuous":
 507            raise ValueError("transformation() is defined for continuous nodes; use cutpoints() for ordinal ones")
 508        params = self._params_for(state)[node]
 509        k_min, k_max = self._minmax(node)
 510        with jax.default_device(self.device):
 511            enc = to_device(st.encode_frame(df, with_target=False))
 512            int_out, shift_sum, _ = node_forward(st, params, enc)
 513            thetas = transform_intercepts_continuous(int_out)
 514            if x is None:
 515                y = jnp.asarray(df[node].to_numpy(), jnp.float32)
 516                return np.asarray(h_extrapolated(thetas, y, k_min, k_max) + shift_sum)
 517            grid = jnp.asarray(np.asarray(x, np.float32))
 518            f = jax.vmap(lambda th, s: h_extrapolated(jnp.broadcast_to(th, (grid.shape[0], th.shape[0])), grid, k_min, k_max) + s)
 519            return np.asarray(f(thetas, shift_sum))

 521    def cutpoints(self, node, df, state="best"):
 522        """Shifted cut-points ``theta_k - eta(pa)`` of an ordinal node, shape ``(len(df), K-1)``."""
 523        st = self.structures[node]
 524        params = self._params_for(state)[node]
 525        with jax.default_device(self.device):
 526            enc = to_device(st.encode_frame(df, with_target=False))
 527            int_out, shift_sum, _ = node_forward(st, params, enc)
 528            return np.asarray(ordinal_cutpoints(int_out) - shift_sum[:, None])

 530    def shift_term(self, node, term=0, x=None, state="best"):
 531        """Evaluate one shift network of ``node`` on input ``x`` (``(n, n_features)`` or 1d).
 532
 533        ``term`` is the index of the shift group or the name of (one of) its parents.
 534        Example: inspect a learned complex shift ``cs(x2)`` of ``x3`` on a grid::
 535
 536            grid = np.linspace(-1, 1, 200)
 537            plt.plot(grid, model.shift_term("x3", "x2", grid))
 538        """
 539        st = self.structures[node]
 540        if isinstance(term, str):
 541            idx = [i for i, g in enumerate(st.shifts) if term in g.parents]
 542            if not idx:
 543                raise KeyError(f"'{term}' is not a shift input of '{node}'")
 544            term = idx[0]
 545        g = st.shifts[term]
 546        x = np.asarray(x, np.float32)
 547        if x.ndim == 1:
 548            x = x[:, None]
 549        p = self._params_for(state)[node]["shifts"][term]
 550        return np.asarray(get_model_spec(g.model_name).apply(p, jnp.asarray(x)))[:, 0]

 642    def sample(self, do_interventions=None, predefined_latent_samples_df=None, **kwargs):
 643        """Sample from the fitted SCM.
 644
 645        * observational: ``model.sample(number_of_samples=10_000)``
 646        * interventional: ``model.sample(do_interventions={"x1": 0.5})``
 647        * counterfactual: ``u = model.get_latent(obs_df)`` then
 648          ``model.sample(do_interventions={...}, predefined_latent_samples_df=u)``
 649
 650        Options: ``number_of_samples`` (10 000), ``number_of_counterfactual_samples`` (10 000, draws
 651        per observation for ordinal latent intervals), ``seed``, ``state`` ('best'),
 652        ``use_initial_weights_for_sampling``, ``save`` (write ``sampling/sampled.npy`` per node),
 653        ``verbose``. Returns ``(sampled_by_node, latents_by_node)`` with numpy arrays.
 654        """
 655        s = {
 656            "number_of_samples": 10_000,
 657            "number_of_counterfactual_samples": 10_000,
 658            "seed": None,
 659            "state": "best",
 660            "use_initial_weights_for_sampling": False,
 661            "save": True,
 662            "delete_all_previously_sampled": True,
 663            "verbose": self.verbose,
 664            "debug": False,
 665            "batch_size": None,
 666            "device": None,
 667        }
 668        unknown = set(kwargs) - set(s)
 669        if unknown:
 670            raise ValueError(f"[sample] Unknown parameter(s): {', '.join(sorted(unknown))}")
 671        s.update(kwargs)
 672        state = "init" if s["use_initial_weights_for_sampling"] else s["state"]
 673        params = self._params_for(state)
 674        mm = {n: self._minmax(n) for n in self.order if self.structures[n].target == "continuous"}
 675        if getattr(self, "_sampler", None) is None or self._sampler.minmax != mm:
 676            self._sampler = DagSampler(self.structures, self.order, params, mm)
 677        sampler = self._sampler
 678        sampler.params = params
 679        with jax.default_device(self.device if s["device"] is None else _resolve_device(s["device"])):
 680            sampled, latents, draws = sampler.sample(
 681                s["number_of_samples"], do=do_interventions, latents_df=predefined_latent_samples_df,
 682                n_cf=s["number_of_counterfactual_samples"], seed=s["seed"], verbose=s["verbose"] or s["debug"],
 683            )
 684        self.last_counterfactual_draws = draws
 685        if s["save"] and self.experiment_dir:
 686            for n in self.order:
 687                d = os.path.join(self._node_dir(n), "sampling")
 688                if s["delete_all_previously_sampled"] and os.path.isdir(d):
 689                    shutil.rmtree(d)
 690                os.makedirs(d, exist_ok=True)
 691                if isinstance(sampled[n], dict):
 692                    np.savez(os.path.join(d, "hist.npz"), **sampled[n])
 693                else:
 694                    np.save(os.path.join(d, "sampled.npy"), sampled[n])
 695                np.save(os.path.join(d, "latents.npy"), latents[n])
 696                if n in draws:
 697                    np.save(os.path.join(d, "draws.npy"), draws[n])
 698        return sampled, latents

 950def apply_warm_start(params, st, thetas_R, shift_coefs=None):
 951    """Place R estimates into the last intercept layer (first input channel) and linear shifts."""
 952    thetas_R = np.asarray(thetas_R, np.float64).ravel()
 953    if st.target == "continuous":
 954        tilde = inverse_transform_intercepts_continuous(thetas_R)
 955    else:
 956        tilde = inverse_transform_intercepts_ordinal(thetas_R)
 957    if tilde.size != st.n_thetas:
 958        raise ValueError(f"R returned {tilde.size} intercepts, expected {st.n_thetas}")
 959    params = jax.tree_util.tree_map(lambda x: x, params)
 960    last = dict(params["int"][-1])
 961    w = np.zeros(last["w"].shape, np.float32)
 962    w[0, :] = tilde
 963    last["w"] = jnp.asarray(w)
 964    if "b" in last:
 965        last["b"] = jnp.zeros_like(last["b"])
 966    params["int"] = list(params["int"][:-1]) + [last]
 967    if shift_coefs is not None and len(shift_coefs) == len(params["shifts"]):
 968        new = []
 969        for sp, c in zip(params["shifts"], shift_coefs):
 970            layer = dict(sp[-1])
 971            w = np.zeros(layer["w"].shape, np.float32)
 972            w[0, 0] = float(c)
 973            layer["w"] = jnp.asarray(w)
 974            new.append(list(sp[:-1]) + [layer])
 975        params["shifts"] = new
 976    return params

313    def fit(self, train_data, val_data=None, **kwargs):
314        """Train all (or ``train_list``) node models jointly.
315
316        Main options: ``epochs``, ``learning_rate``, ``batch_size`` (each a scalar, or for
317        ``epochs``/``learning_rate`` a ``{node: value}`` dict), ``optimizer`` (an optax
318        ``GradientTransformation`` replacing Adam), ``seed``, ``verbose``, ``resume`` (continue
319        from ``last_params.npz`` and the stored histories, default True).
320        Defaults come from ``cfg.conf_dict['training']`` when present (written by the
321        configuration editor), otherwise from ``DEFAULTS_FIT``.
322        Arguments specific to torch DataLoaders (``num_workers`` ...) are accepted and ignored.
323        """
324        ignored = set(kwargs) & _IGNORED_FIT_KWARGS
325        for k in ignored:
326            kwargs.pop(k)
327        unknown = set(kwargs) - set(self.DEFAULTS_FIT)
328        if unknown:
329            raise ValueError(f"[fit] Unknown parameter(s): {', '.join(sorted(unknown))}")
330        cfg_training = {k: v for k, v in (self.cfg.conf_dict.get("training") or {}).items() if k in self.DEFAULTS_FIT}
331        s = {**self.DEFAULTS_FIT, **cfg_training, **kwargs}
332        self.verbose, self.debug = bool(s["verbose"]), bool(s["debug"])
333        if "device" in kwargs:
334            self.device = _resolve_device(s["device"])
335
336        train_df = self._as_frame(train_data)
337        val_df = self._as_frame(val_data)
338        nodes = [n for n in self.order if n in (s["train_list"] or self.order)]
339        per_node = lambda v, n: v[n] if isinstance(v, dict) else v
340
341        self.load_or_compute_minmax(train_df, use_existing=False, write=True)
342
343        # ---- resume
344        hist_old, start, remaining = {}, {}, {}
345        for n in nodes:
346            hist_old[n] = self._read_histories(n) if s["resume"] else None
347            last = self._load_state(n, "last") if (s["resume"] and self.experiment_dir) else None
348            if hist_old[n] and last is not None:
349                self.params[n] = last
350                best = self._load_state(n, "best")
351                self.best_params[n] = best if best is not None else last
352                start[n] = len(hist_old[n]["train"])
353            else:
354                hist_old[n] = None
355                start[n] = 0
356            remaining[n] = max(0, int(per_node(s["epochs"], n)) - start[n])
357            if self.verbose:
358                if remaining[n] == 0:
359                    print(f"[INFO] Node {n} already trained for {start[n]} epochs. Skipping.")
360                elif start[n]:
361                    print(f"[INFO] Node {n}: resuming at epoch {start[n]}.")
362        nodes = [n for n in nodes if remaining[n] > 0]
363        if not nodes:
364            return {} if s["return_history"] else None
365
366        if self.verbose:
367            print(f"[INFO] Training {len(nodes)} node(s) jointly on {self.device} "
368                  f"({len(train_df)} rows, batch size {s['batch_size']})")
369            for n in nodes:
370                print(f"       {self.structures[n].formula()}")
371
372        best_score = {}
373        for n in nodes:
374            h = hist_old[n]
375            if h:
376                vals = [v for v in (h["val"] if any(np.isfinite(h["val"])) else h["train"]) if np.isfinite(v)]
377                best_score[n] = min(vals) if vals else np.inf
378            else:
379                best_score[n] = np.inf
380
381        seed = s["seed"] if s["seed"] is not None else int(self.settings.get("seed") or 0) + sum(start.values())
382        with jax.default_device(self.device):
383            train_enc = self._encode(train_df, nodes)
384            val_enc = self._encode(val_df, nodes) if val_df is not None and len(val_df) else None
385            trainer = JointTrainer(
386                {n: self.structures[n] for n in nodes}, nodes, {n: self._minmax(n) for n in nodes if self.structures[n].target == "continuous"},
387                s["batch_size"], {n: per_node(s["learning_rate"], n) for n in nodes}, optimizer=s["optimizer"], seed=seed,
388            )
389            p_sub = {n: self.params[n] for n in nodes}
390            b_sub = {n: self.best_params.get(n, self.params[n]) for n in nodes}
391            p_sub, b_sub, best_score, hist = trainer.fit(
392                p_sub, train_enc, val_enc, remaining, best_params=b_sub, best_score=best_score,
393                verbose=self.verbose, epochs_per_call=s["epochs_per_call"],
394            )
395        for n in nodes:
396            self.params[n] = p_sub[n]
397            self.best_params[n] = b_sub[n]
398        self._write_fit_outputs(nodes, hist, hist_old, start, remaining, s)
399        if self.verbose:
400            print(f"[INFO] Training finished in {hist['seconds']:.1f}s")
401        if s["return_history"]:
402            return {n: (self._read_histories(n) or {}) for n in nodes}
403        return None

504        the grid for each row's parents; otherwise ``h`` at the observed ``df[node]``.
505        """
506        st = self.structures[node]
507        if st.target != "continuous":
508            raise ValueError("transformation() is defined for continuous nodes; use cutpoints() for ordinal ones")
509        params = self._params_for(state)[node]
510        k_min, k_max = self._minmax(node)
511        with jax.default_device(self.device):
512            enc = to_device(st.encode_frame(df, with_target=False))
513            int_out, shift_sum, _ = node_forward(st, params, enc)
514            thetas = transform_intercepts_continuous(int_out)
515            if x is None:
516                y = jnp.asarray(df[node].to_numpy(), jnp.float32)
517                return np.asarray(h_extrapolated(thetas, y, k_min, k_max) + shift_sum)
518            grid = jnp.asarray(np.asarray(x, np.float32))
519            f = jax.vmap(lambda th, s: h_extrapolated(jnp.broadcast_to(th, (grid.shape[0], th.shape[0])), grid, k_min, k_max) + s)
520            return np.asarray(f(thetas, shift_sum))

523        """Shifted cut-points ``theta_k - eta(pa)`` of an ordinal node, shape ``(len(df), K-1)``."""

643    def sample(self, do_interventions=None, predefined_latent_samples_df=None, **kwargs):
644        """Sample from the fitted SCM.
645
646        * observational: ``model.sample(number_of_samples=10_000)``
647        * interventional: ``model.sample(do_interventions={"x1": 0.5})``
648        * counterfactual: ``u = model.get_latent(obs_df)`` then
649          ``model.sample(do_interventions={...}, predefined_latent_samples_df=u)``
650
651        Options: ``number_of_samples`` (10 000), ``number_of_counterfactual_samples`` (10 000, draws
652        per observation for ordinal latent intervals), ``seed``, ``state`` ('best'),
653        ``use_initial_weights_for_sampling``, ``save`` (write ``sampling/sampled.npy`` per node),
654        ``verbose``. Returns ``(sampled_by_node, latents_by_node)`` with numpy arrays.
655        """
656        s = {
657            "number_of_samples": 10_000,
658            "number_of_counterfactual_samples": 10_000,
659            "seed": None,
660            "state": "best",
661            "use_initial_weights_for_sampling": False,
662            "save": True,
663            "delete_all_previously_sampled": True,
664            "verbose": self.verbose,
665            "debug": False,
666            "batch_size": None,
667            "device": None,
668        }
669        unknown = set(kwargs) - set(s)
670        if unknown:
671            raise ValueError(f"[sample] Unknown parameter(s): {', '.join(sorted(unknown))}")
672        s.update(kwargs)
673        state = "init" if s["use_initial_weights_for_sampling"] else s["state"]
674        params = self._params_for(state)
675        mm = {n: self._minmax(n) for n in self.order if self.structures[n].target == "continuous"}
676        if getattr(self, "_sampler", None) is None or self._sampler.minmax != mm:
677            self._sampler = DagSampler(self.structures, self.order, params, mm)
678        sampler = self._sampler
679        sampler.params = params
680        with jax.default_device(self.device if s["device"] is None else _resolve_device(s["device"])):
681            sampled, latents, draws = sampler.sample(
682                s["number_of_samples"], do=do_interventions, latents_df=predefined_latent_samples_df,
683                n_cf=s["number_of_counterfactual_samples"], seed=s["seed"], verbose=s["verbose"] or s["debug"],
684            )
685        self.last_counterfactual_draws = draws
686        if s["save"] and self.experiment_dir:
687            for n in self.order:
688                d = os.path.join(self._node_dir(n), "sampling")
689                if s["delete_all_previously_sampled"] and os.path.isdir(d):
690                    shutil.rmtree(d)
691                os.makedirs(d, exist_ok=True)
692                if isinstance(sampled[n], dict):
693                    np.savez(os.path.join(d, "hist.npz"), **sampled[n])
694                else:
695                    np.save(os.path.join(d, "sampled.npy"), sampled[n])
696                np.save(os.path.join(d, "latents.npy"), latents[n])
697                if n in draws:
698                    np.save(os.path.join(d, "draws.npy"), draws[n])
699        return sampled, latents
